class Bien {
private:
	int codigo;
	char tipo;
	double valor;
	int perdida;

public:	
	virtual void setPerdida(int valorPerdida){
		this->perdida = valorPerdida;
	}
	
	virtual void amortizar(){
		double nuevoValor;
		nuevoValor = this->valor - (this->valor * perdida);
		setPerdida(nuevoValor);	
	}
}

class Terreno : public Bien {

}

		if(valor<valorMin){

class Avion : public Viaje {
private:
	double extraPrecio = 0.2;
	double extraHoras = 2;

public:
	double getPrecio() override {
		return precio + (precio * extraPrecio);
	}
}

class Colectivo : public Viaje {
private:
	float distancia;
	int costoKm = 8;
public:
	double getPrecio() {
		return distancia * costoKm;
	}

}

class Dual : public Viaje {
private:
	Avion* avion;
	Colectivo* cole;

	double getPrecio () override {
		return (avion->getPrecio) + (cole->getPrecio);
	}
