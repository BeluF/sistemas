\section{Límites}

Permite analizar el comportamiento de una función cuando se acerca a un punto, sin necesidad de alcanzarlo.

\subsection{Definiciones Formales e Intuitivas}

\begin{itemize}
    \item \textbf{Intuitiva:} El límite de $f(x)$ cuando $x$ tiende a $a$ es $L$ si podemos hacer que los valores de $f(x)$ estén arbitrariamente cerca de $L$ tomando $x$ suficientemente cerca de $a$ (pero $x \neq a$).
    \item \textbf{Formal (Definición $\epsilon-\delta$):}
    Para cualquier $\epsilon > 0$, existe un $\delta > 0$ tal que:
    \[
    0 < |x - a| < \delta \implies |f(x) - L| < \epsilon
    \]
\end{itemize}

\subsection{Límites laterales}
\begin{itemize}
    \item \textbf{Límite por la izquierda:} $\lim_{x \to a^-} f(x) = L$ si $f(x)$ se acerca a $L$ cuando $x$ se acerca a $a$ desde valores menores.
    \item \textbf{Límite por la derecha:} $\lim_{x \to a^+} f(x) = L$ si $f(x)$ se acerca a $L$ cuando $x$ se acerca a $a$ desde valores mayores.
\end{itemize}

\subsection{Límites infinitos}
\begin{itemize}
    \item \textbf{Límite infinito:} $\lim_{x \to a} f(x) = \infty$ o $\lim_{x \to a} f(x) = -\infty$ significa que los valores de $f(x)$ pueden volverse muy grandes (positivos o negativos) cuando $x$ se acerca a $a$.
    \item \textbf{Límite al infinito:} Sea la función $f$ definida en el intervalo $(a, \infty)$ o $(-\infty, b)$, entonces
    \begin{itemize}
        \item \textbf{Límite superior:} $\lim_{x \to \infty} f(x) = L$ significa que los valores de $f(x)$ se acercan a $L$ cuando $x$ toma un valor muy grande positivo.
        \item \textbf{Límite inferior: } $\lim_{x \to -\infty} f(x) = L$ significa que los valores de $f(x)$ se acercan a $L$ cuando $x$ toma un valor muy grande negativo.
    \end{itemize}
\end{itemize}

\subsection{Asíntota vertical y horizontal}
\begin{itemize}
    \item \textbf{Asíntota vertical:} La línea $x = a$ es una \textbf{asíntota vertical} de la curva $y = f(x)$ si $\lim_{x \to a^-} f(x) = \pm \infty$ o $\lim_{x \to a^+} f(x) = \pm \infty$.
    \item \textbf{Asíntota horizontal:} La línea $y = L$ es una \textbf{asíntota horizontal} de de la curva $y = f(x)$ si $\lim_{x \to \infty} f(x) = L$ o $\lim_{x \to -\infty} f(x) = L$.
\end{itemize}

\subsection{Estrategias de Resolución}
\begin{enumerate}
    \item \textbf{Sustitución Directa:} Si $f$ es continua (polinomios, racionales en su dominio), evaluar $f(a)$: $\lim_{x \to a} f(x) = f(a)$.
    \item \textbf{Forma Indeterminada $\frac{0}{0}$:}
    \begin{itemize}
        \item \textit{Factorización:} Simplificar términos comunes.
        \item \textit{Racionalización:} Multiplicar por el conjugado si hay raíces.
    \end{itemize}
    \item \textbf{Límites al Infinito:} Dividir numerador y denominador por la mayor potencia de $x$ del denominador.
\end{enumerate}

\subsection{Teoremas Fundamentales}
\begin{enumerate}
    \item \textbf{Existencia del límite:} $\lim_{x \to a} f(x) = L \leftrightarrow \lim_{x \to a^-} f(x) = \lim_{x \to a^+} f(x) = L$.
    \item \textbf{Desigualdad de límites:} Si $f(x) \leqslant g(x)$ cuando $x \to a$, entonces $\lim_{x \to a} f(x) \leqslant \lim_{x \to a} g(x)$.
    \item \textbf{Teorema de la Compresión (Sándwich):} Si $f(x) \leqslant g(x) \leqslant h(x)$ cuando $x \to a$ y $\lim_{x\to a} f(x) = \lim_{x\to a} h(x) = L$, entonces $\lim_{x\to a} g(x) = L$.
    \item \textbf{Racionales: } Si $r > 0$ es racional, entonces:
    \[\lim_{x \to \infty} \frac{1}{x^r} = 0 \quad \text{y si $x^r$ está definida $\forall x$} \Rightarrow \quad \lim_{x \to -\infty} \frac{1}{x^r} = 0\]
\end{enumerate}

\subsection{Límites infinitos al infinito}
\[ \lim_{x \to \pm \infty} f(x) = \pm \infty \] significa que los valores de $f(x)$ pueden volverse arbitrariamente grandes cuando $x$ toma valores muy grandes.

\section{Continuidad}

\subsection{Definiciones Formales e Intuitivas}
\begin{itemize}
    \item \textbf{Continuidad en un punto:} Una función $f$ es continua en $a$ si cumple tres condiciones estrictas:
    \begin{enumerate}
        \item $f(a)$ está definida.
        \item $\lim_{x \to a} f(x)$ existe.
        \item $\lim_{x \to a} f(x) = f(a)$.
    \end{enumerate}
    \item \textbf{Continuidad por los laterales:}
        \begin{itemize}
            \item $\lim_{x \to a^-} f(x) = f(a)$ (continua por la izquierda).
            \item $\lim_{x \to a^+} f(x) = f(a)$ (continua por la derecha).
        \end{itemize}
    \item \textbf{Tipos de Discontinuidad:} Removible (límite existe pero no coincide), De Salto (límites laterales distintos), Infinita (asíntota vertical).
\end{itemize}

\subsection{Continuidad en Intervalos}
Una función $f$ es continua en un intervalo $I$ si es continua en cada punto de $I$.

\subsection{Teoremas Fundamentales}
\begin{itemize}
    \item \textbf{Operaciones con funciones continuas:} Si $f$ y $g$ son continuas en $a$, y existe una constante $c$, entonces $f+g$, $f-g$, $fg$, $cf$ y $\frac{f}{g}$ (si $g(a) \neq 0$) son continuas en $a$.
    \item \textbf{Teorema del Valor Intermedio (TVI):} Si $f$ es continua en el intervalo cerrado $[a, b]$ y sea $N$ cualquier número entre $f(a)$ y $f(b)$, donde $f(a) \leqslant f(b) \Rightarrow \exists c \in (a, b) : f(c) = N$. Base para hallar raíces.
\end{itemize}

\subsection{Propiedades de Continuidad}
\begin{table}[h!]
    \centering
    \renewcommand{\arraystretch}{1.2}
    \begin{tabular}{l l}
        \toprule
        \textbf{Tipo de Función} & \textbf{Regla} \\
        \midrule
        Polinomios & Continuos en todo $\mathbb{R}$ \\
        Racionales & Continuas en su dominio (denominador $\neq 0$) \\
        Trigonométricas & Continuas en su dominio \\
        Composición & $\lim_{x \to a} f(g(x)) = f(\lim_{x \to a} g(x))$ \\
        Composición & Si $g$ cont. en $a$ y $f$ cont. en $g(a) \implies (f \circ g) = f(g(x))$ cont. en $a$ \\
        \bottomrule
    \end{tabular}
\end{table}

\subsection{Estrategias de Resolución}
\begin{itemize}
    \item Verificar las 3 condiciones de la definición para puntos sospechosos.
    \item Para funciones a trozos, asegurar que los límites laterales coincidan en los puntos de cambio.
\end{itemize}

\section{Derivadas}

\subsection{Definiciones Formales e Intuitivas}
\begin{itemize}
    \item \textbf{Definición Formal:} Límite del cociente incremental:
    \[ f'(x) = \lim_{h \to 0} \frac{f(x+h) - f(x)}{h} \]
    \item \textbf{Interpretación:} Pendiente de la recta tangente y razón de cambio instantánea.
\end{itemize}

\subsection{Derivabilidad}
\begin{itemize}
    \item \textbf{Derivabilidad en un punto:} $f$ es derivable en $a$ si $f'(a)$ existe.
    \item \textbf{Derivabilidad en un intervalo:} $f$ es derivable en $I$ si es derivable en cada punto de $I$.
    \item \textbf{Relación con la continuidad:} Si $f$ es derivable en $a$, entonces es continua en $a$. La recíproca no siempre es cierta.
\end{itemize}

\begin{tcolorbox}[title=No Derivabilidad]
    Si la función tiene un una esquina, una discontinuidad o una tangente vertical en $x=a$, entonces no es derivable en $a$.
\end{tcolorbox}

\subsection{Reglas de Derivación}
\begin{table}[h!]
    \centering
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{l l}
        \toprule
        \textbf{Regla} & \textbf{Expresión Matemática} \\
        \midrule
        Producto & $(fg)' = f g' + g f'$ \\
        Cociente & $\left(\frac{f}{g}\right)' = \frac{g f' - f g'}{g^2}$ \\
        Cadena & $\frac{d}{dx}[f(g(x))] = f'(g(x)) \cdot g'(x)$ \\
        Exponencial/Log & $(e^x)' = e^x$, \quad $(\ln x)' = \frac{1}{x}$ \\
        Potencia + Cadena & $\frac{d}{dx}[u(x)]^n = n[u(x)]^{n-1} \cdot u'(x)$ \\
        \bottomrule
    \end{tabular}
\end{table}

\begin{tcolorbox}[title=Derivadas Trigonométricas]
    En funciones trigonométricas, los signos menos van con las derivadas de las "cofunciones" (coseno, cosecante, cotangente).
\end{tcolorbox}

\subsection{TVM}
\textbf{Teorema del Valor Medio:} Si $f$ es continua en $[a,b]$ y derivable en $(a,b)$, existe $c$ tal que $f'(c) = \frac{f(b)-f(a)}{b-a}$. Si $f'(x) = 0$ en $(a,b)$, entonces $f$ es constante en $(a,b)$.

\subsection{Estrategias de Resolución}
\begin{itemize}
    \item \textbf{Derivación Implícita:} Para ecuaciones $F(x,y)=0$, derivar ambos lados respecto a $x$ y luego resolver la ecuación restante para $y'$.
    \item \textbf{Optimización:} Hallar puntos críticos ($f'=0$), evaluar en extremos del intervalo o usar criterio de $2^{da}$ derivada.
\end{itemize}

\subsection{Derivadas y gráficas}
\begin{itemize}
    \item \textbf{Crecimiento}: Si $f'(x) > 0$ en un intervalo, $f$ es \textbf{creciente} en ese intervalo.
    \item \textbf{Decrecimiento}: Si $f'(x) < 0$ en un intervalo, $f$ es \textbf{decreciente} en ese intervalo.
\end{itemize}

\begin{tcolorbox}[title=Prueba de la primera y segunda derivada]
    Las pruebas de la primera y segunda derivada son herramientas para determinar la naturaleza de los puntos críticos de una función. \newline
    \textbf{Primera derivada:}
\begin{itemize}
    \item Si $f'(x)$ cambia de positivo a negativo en $c$, entonces $f$ tiene un \textbf{máximo local} en $c$.
    \item Si $f'(x)$ cambia de negativo a positivo en $c$, entonces $f$ tiene un \textbf{mínimo local} en $c$.
\end{itemize}
    \textbf{Segunda derivada:}
    \begin{itemize}
    \item Si $f''(c) > 0$, entonces $f$ tiene un \textbf{mínimo local} en $c$.
    \item Si $f''(c) < 0$, entonces $f$ tiene un \textbf{máximo local} en $c$.
\end{itemize}
\end{tcolorbox}

\subsection{Concavidad y Puntos de Inflexión}
\begin{itemize}
    \item Si $f''(x) > 0$ en un intervalo, $f$ es \textbf{cóncava hacia arriba} en ese intervalo (la gráfica de $f$ está arriba de sus tangentes).
    \item Si $f''(x) < 0$ en un intervalo, $f$ es \textbf{cóncava hacia abajo} en ese intervalo (la gráfica de $f$ está abajo de sus tangentes).
\end{itemize}

\begin{tcolorbox}
Si $f''$ cambia de signo (concavidad) en $c$, entonces la gráfica de $f$ tiene un \textbf{punto de inflexión} en $c$.
\end{tcolorbox}

\subsection{L'Hôpital}
\textbf{Regla de L'Hôpital:} Si $\lim_{x \to a} f(x) = 0$ y $\lim_{x \to a} g(x) = 0$ o ambos límites son $\pm \infty$, entonces:
\[ \lim_{x \to a} \frac{f(x)}{g(x)} = \lim_{x \to a} \frac{f'(x)}{g'(x)} \]
si el límite del lado derecho existe o es $\pm \infty$.

\subsubsection{Formas Indeterminadas}
En cualquiera de los siguientes casos (salvo los iniciales), debe llevarse la indeterminación a $\frac{0}{0}$ o $\frac{\infty}{\infty}$ para poder aplicar la regla de L'Hôpital:
\begin{multicols}{3}
\begin{itemize}
    \item $\frac{0}{0}$
    \item $\frac{\infty}{\infty}$
    \item $0 \cdot \infty$
    \item $\infty - \infty$
    \item $0^0$
    \item $1^\infty$
    \item $\infty^0$
\end{itemize}
\end{multicols}

\section{Integrales}

\subsection{Definición formal}
Una \textbf{integral definida} es el límite de sumas de rectángulos. Representa área neta.
    \[ \int_{a}^{b} f(x) \, dx = \lim_{n \to \infty} \sum_{i=1}^{n} f(x_i^*) \Delta x \] para $a < b$.
    \[ \int_{a}^{b} f(x) \, dx = - \int_{b}^{a} f(x) \, dx \] para $a > b$.
    \[ \int_{a}^{b} f(x) \, dx = \int_{b}^{a} f(x) \, dx = 0 \] para $a = b$.

\subsection{Integrabilidad}
\textbf{Criterio de Integrabilidad de Riemann:} Si $f$ es continua en $[a,b]$ o si $f$ tiene un número finito de discontinuidades en $[a,b]$, $f$ es integrable.
\newline Si $f$ es integrable en $[a,b]$, entonces 
\[ \int_{a}^{b} f(x) \, dx = \lim_{n \to \infty} \sum_{i=1}^{n} f(x_i^*) \Delta x \] 
%Agregar deltax y xi*

\subsection{Propiedades de la Integral Definida}
\begin{tcolorbox}[title=Propiedades de la Integral Definida]
    \begin{itemize}
        \item $\int_{a}^{b} c f(x) \, dx = c \int_{a}^{b} f(x) \, dx$
        \item $\int_{a}^{b} [f(x) + g(x)] \, dx = \int_{a}^{b} f(x) \, dx + \int_{a}^{b} g(x) \, dx$
        \item $\int_{a}^{b} c f(x) \, dx = c \int_{a}^{b} f(x) \, dx$
        \item $\int_{a}^{b} [f(x) - g(x)] \, dx = \int_{a}^{b} f(x) \, dx - \int_{a}^{b} g(x) \, dx$
        \item $\int_{a}^{c} f(x) \, dx + \int_{c}^{b} f(x) \, dx = \int_{a}^{b} f(x) \, dx$
    \end{itemize}
\end{tcolorbox}

\subsubsection{Propiedades de comparación integral}
\begin{itemize}
    \item Si $f(x) \geqslant 0$ para $a \leqslant x \leqslant b$, entonces $\int_{a}^{b} f(x) \, dx \geqslant 0$.
    \item Si $f(x) \geqslant g(x)$ para $a \leqslant x \leqslant b$, entonces $\int_{a}^{b} f(x) \, dx \geqslant \int_{a}^{b} g(x) \, dx$.
    \item Si $m \leqslant f(x) \leqslant M$ para $a \leqslant x \leqslant b$, entonces \[ m(b-a) \leqslant \int_{a}^{b} f(x) \, dx \leqslant M(b-a) \]
\end{itemize}

\subsection{Teorema Fundamental del Cálculo}
Sea $f$ continua en $[a,b]$.
    \begin{itemize}
        \item Si $g(x) = \int_{a}^{x} f(t) \, dt$, entonces $g'(x) = f(x)$.
        \item $\int_{a}^{b} f(x) \, dx = F(b) - F(a)$, donde $F$ es cualquier antiderivada de $f$ ($F' = f$).
    \end{itemize}

\subsection{Definición formal}
Una \textbf{integral indefinida} es la colección de todas las antiderivadas de una función $f$. Se denota como:
    \[ \int f(x) \, dx = F(x) + C \]

\begin{tcolorbox}[title=Teorema del cambio neto]
    La integral de una razón de cambio es el cambio neto:
    \[ \int_a^b F'(x) \, dx = F(b) - F(a) \]
\end{tcolorbox}

\begin{tcolorbox}[title=TVM en integrales]
    Si $f$ es continua en $[a,b]$, entonces existe $c \in [a,b]$ tal que:
    \[ \int_{a}^{b} f(x) \, dx = f(c)(b-a) \]
\end{tcolorbox}

\subsection{Estrategias de Resolución}
\begin{enumerate}
    \item \textbf{Sustitución ($u$):} Para integrales indefinidas: $\int f(g(x))g'(x) \, dx = \int f(u) \, du$. Para integrales definidas: $\int_a^b f(g(x))g'(x) \, dx = \int_{g(a)}^{g(b)} f(u) \, du$
    \item \textbf{Por Partes:} $\int u \, dv = uv - \int v \, du$. Para productos (ej. $x e^x$).
    \item \textbf{Simetría:} En $[-a, a]$, si $f$ es impar la integral $\int_{-a}^a f(x) \, dx = 0$; si es par, es $2\int_0^a f(x) \, dx$.
\end{enumerate}

\subsection{Integrales impropias}
Las integrales impropias son aquellas en las que \textbf{el intervalo es infinito} o \textbf{$f$ tiene una discontinuidad infinita}.

\subsubsection{Integrales impropias de tipo 1: \textbf{intervalos infinitos}}
\begin{itemize}
    \item Si $\int_{a}^{t} f(x) \, dx$ existe para todo número $t \geqslant a$, entonces
    \[ \int_{a}^{\infty} f(x) \, dx = \lim_{t \to \infty} \int_{a}^{t} f(x) \, dx \]
    siempre que el límite exista.

    \item Si $\int_{t}^{b} f(x) \, dx$ existe para todo número $t \leqslant b$, entonces
    \[ \int_{-\infty}^{b} f(x) \, dx = \lim_{t \to -\infty} \int_{t}^{b} f(x) \, dx \]
    siempre que el límite exista.

    \begin{tcolorbox}
        Las integrales impropias $\int_{a}^{\infty} f(x) \, dx$ y $\int_{-\infty}^{b} f(x) \, dx$ \textbf{convergen} si los límites existen (son finitos); \textbf{divergen} si los límites no existen (son infinitos o no definidos).
    \end{tcolorbox}

    \item Si $\int_{-\infty}^{\infty} f(x) \, dx$ existe, entonces
    \[ \int_{-\infty}^{\infty} f(x) \, dx = \int_{-\infty}^{a} f(x) \, dx + \int_{a}^{\infty} f(x) \, dx \]
    para algún número real $a$.
\end{itemize}

\subsubsection{Integrales impropias de tipo 2: \textbf{discontinuidades infinitas}}
\begin{itemize}
    \item Si $f$ es continua en $[a,b)$ pero tiene una discontinuidad infinita en $x=b$, entonces
    \[ \int_{a}^{b} f(x) \, dx = \lim_{t \to b^-} \int_{a}^{t} f(x) \, dx \]
    siempre que el límite exista.

    \item Si $f$ es continua en $(a,b]$ pero tiene una discontinuidad infinita en $x=a$, entonces
    \[ \int_{a}^{b} f(x) \, dx = \lim_{t \to a^+} \int_{t}^{b} f(x) \,dx \]
    siempre que el límite exista.

    \begin{tcolorbox}
        La integral impropia $\int_{a}^{b} f(x) \, dx$ \textbf{converge} si el límite existe (es finito); \textbf{diverge} si el límite no existe (es infinito o no definido).
    \end{tcolorbox}

    \item Si $f$ tiene una discontinuidad infinita en $x=c$, donde $a < c < b$ y ambas $\int_{a}^{c} f(x) \, dx$ y $\int_{c}^{b} f(x) \, dx$ son \textbf{convergentes}, entonces
    \[ \int_{a}^{b} f(x) \, dx = \int_{a}^{c} f(x) \, dx + \int_{c}^{b} f(x) \, dx \]
\end{itemize}

\subsubsection{Comparación para las integrales impropias}
Si $f$ y $g$ son funciones continuas en $[a, \infty)$ y $f(x) \geqslant g(x) \geqslant 0$ para todo $x \geqslant a$, entonces:
\begin{itemize}
    \item Si $\int_{a}^{\infty} f(x) \,dx$ \textbf{converge}, entonces $\int_{a}^{\infty} g(x) \, dx$ \textbf{converge}.
    \item Si $\int_{a}^{\infty} g(x) \, dx$ \textbf{diverge}, entonces $\int_{a}^{\infty} f(x) \, dx$ \textbf{diverge}.
\end{itemize}
