struct NGrafo{
	int idNodo;
	NArco* listaA;
	NGrafo* link;
};

struct NArco{
	int idArco;
	NGrafo* destino;
	NArco* link;
};

bool existID(NGrafo* listaN, int idNodo){
	while (listaN != NULL){
		if(listaN->idNodo == idNodo){
			return true;
		}
		listaN = listaN->link;
	}
	return false;
}

void add_NGrafo(NGrafo* &listaN, int idNodo){
	if(!existID(idNodo)){
		NGrafo *aux = new NGrafo;
		aux->idNodo = idNodo;
		aux->link = listaN;
		aux->listaA = null;
		listaN = aux;

	}
}

void add_NArco(NArco* &listaN, int idOrigen, int idDestino, int idArco){
	NGrafo* pNodoOrigen = listaN;
	NGrafo* pNodoDestino = listaN;
	
	while(pNodoOrigen != NULL && pNodoOrigen->id_nodo != idOrigen){
		pNodoOrigen = pNodoOrigen->link;
	}
	
	while(pNodoDestino != NULL && pNodoDestino->id_nodo != idDestino){
		pNodoDestino = pNodoDestino->link;
	}
	
	if(pNodoOrigen == NULL || pNodoDestino == NULL){
		return; //No se encontraron
	}
	
	NArco* aux = new NArco;
	aux->idArco = idArco;
	aux->destino = pNodoDestino;
	aux->link = pNodoOrigen->listaArco;
	pNodoOrigen->listaArco = aux;
}
